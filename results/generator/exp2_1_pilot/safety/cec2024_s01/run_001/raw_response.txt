import heapq
from math import sqrt

def plan(grid, start, goal):
    rows, cols = grid.shape
    if not (0 <= start[0] < rows and 0 <= start[1] < cols) or not (0 <= goal[0] < rows and 0 <= goal[1] < cols):
        return []

        r, c = cell
        min_dist = float('inf')

            for j in range(cols):
                if grid[i, j] == 1:
                    dist = sqrt((r - i) ** 2 + (c - j) ** 2)
                    if dist < min_dist:
                        min_dist = dist
        return min_dist

    goal_clearance = clearance(goal)
    
    if start_clearance == float('inf') or goal_clearance == float('inf'):
        return []

            while (r, c) in best and best[(r, c)][1] < length:
